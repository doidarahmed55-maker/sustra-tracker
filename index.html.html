⚙️ إعداد Supabase — مرة واحدة بس

عشان الـ app يشتغل أونلاين مع Real-time sync، محتاج ترفع الداتابيز على Supabase مجاناً. اتبع الخطوات دي:

الخطوة 1 — إنشاء Project

روح supabase.com → New project → اختار اسم وباسورد → انتظر دقيقتين

الخطوة 2 — إنشاء الجداول

افتح SQL Editor في Supabase وانسخ والصق الكود ده واضغط Run:

-- إنشاء جدول اللوحات
CREATE TABLE IF NOT EXISTS drawings (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  num integer,
  plot text,
  zone text,
  level text,
  type text,
  name text NOT NULL,
  content text,
  del date,
  ns boolean DEFAULT false,
  by_eng text,
  link text,
  paid text DEFAULT 'unpaid',
  notes text,
  r1r date, r1p date, r1nc boolean DEFAULT false,
  r2r date, r2p date, r2nc boolean DEFAULT false,
  r3r date, r3p date, r3nc boolean DEFAULT false,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- تفعيل Real-time
ALTER TABLE drawings REPLICA IDENTITY FULL;

-- سماح للمستخدمين المسجلين بالقراءة والكتابة
ALTER TABLE drawings ENABLE ROW LEVEL SECURITY;
CREATE POLICY "auth_all" ON drawings FOR ALL TO authenticated USING (true) WITH CHECK (true);

-- إنشاء جدول أسماء المهندسين
CREATE TABLE IF NOT EXISTS engineers (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text UNIQUE NOT NULL
);
ALTER TABLE engineers ENABLE ROW LEVEL SECURITY;
CREATE POLICY "auth_all_eng" ON engineers FOR ALL TO authenticated USING (true) WITH CHECK (true);
